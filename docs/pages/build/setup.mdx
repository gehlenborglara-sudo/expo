// App.js
import React, { useState, useEffect } from 'react';
import { StyleSheet, View, Text, TouchableOpacity, Modal, TextInput, Button, Platform } from 'react-native';
import MapView, { Marker } from 'react-native-maps';
import * as Location from 'expo-location';

export default function App() {
  const [region, setRegion] = useState({
    latitude: 52.52,
    longitude: 13.405,
    latitudeDelta: 0.05,
    longitudeDelta: 0.05,
  });

  const [parkings, setParkings] = useState([
    { id: '1', coord: { latitude: 52.52, longitude: 13.40 }, price: 2.5, status: 'free', title: 'Privatparkplatz A' },
    { id: '2', coord: { latitude: 52.515, longitude: 13.41 }, price: 3.0, status: 'occupied', title: 'Garage B' },
    { id: '3', coord: { latitude: 52.518, longitude: 13.395 }, price: 2.0, status: 'free', title: 'Parkplatz C' },
  ]);

  const [selectedParking, setSelectedParking] = useState(null);
  const [bookingModalVisible, setBookingModalVisible] = useState(false);
  const [providerModalVisible, setProviderModalVisible] = useState(false);

  // New parking state
  const [newParkingCoord, setNewParkingCoord] = useState(null);
  const [newPrice, setNewPrice] = useState('');
  const [newTitle, setNewTitle] = useState('');

  // Request user location
  useEffect(() => {
    (async () => {
      let { status } = await Location.requestForegroundPermissionsAsync();
      if (status !== 'granted') {
        console.log('Permission to access location was denied');
        return;
      }

      let location = await Location.getCurrentPositionAsync({});
      setRegion({
        latitude: location.coords.latitude,
        longitude: location.coords.longitude,
        latitudeDelta: 0.05,
        longitudeDelta: 0.05,
      });
    })();
  }, []);

  // Handle long press for new parking spot
  const handleLongPress = (e) => {
    setNewParkingCoord(e.nativeEvent.coordinate);
    setProviderModalVisible(true);
  };

  // Add new parking spot
  const addNewParking = () => {
    if (!newPrice || !newParkingCoord || !newTitle) {
      alert('Bitte alle Felder ausfüllen.');
      return;
    }
    const newSpot = {
      id: (parkings.length + 1).toString(),
      coord: newParkingCoord,
      price: parseFloat(newPrice),
      status: 'free',
      title: newTitle,
    };
    setParkings([...parkings, newSpot]);
    setProviderModalVisible(false);
    setNewParkingCoord(null);
    setNewPrice('');
    setNewTitle('');
  };

  // Dummy booking confirm
  const confirmBooking = () => {
    alert(`Parkplatz "${selectedParking.title}" gebucht!`);
    setBookingModalVisible(false);
    setSelectedParking(null);
  };

  return (
    <View style={styles.container}>
      <MapView
        style={styles.map}
        region={region}
        onRegionChangeComplete={setRegion}
        showsUserLocation={true}
        onLongPress={handleLongPress}
      >
        {parkings.map((p) => (
          <Marker
            key={p.id}
            coordinate={p.coord}
            pinColor={p.status === 'free' ? 'green' : 'red'}
            title={p.title}
            onPress={() => setSelectedParking(p)}
          >
            <View style={styles.marker}>
              <Text style={{ color: 'white', fontWeight: 'bold' }}>{p.price} €/h</Text>
            </View>
          </Marker>
        ))}
      </MapView>

      {/* Booking Modal */}
      <Modal visible={bookingModalVisible} transparent animationType="slide">
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>Parkplatz buchen</Text>
            {selectedParking && (
              <>
                <Text>{selectedParking.title}</Text>
                <Text>Preis: {selectedParking.price} €/h</Text>
                <Button title="Jetzt buchen" onPress={confirmBooking} />
                <Button title="Abbrechen" onPress={() => setBookingModalVisible(false)} />
              </>
            )}
          </View>
        </View>
      </Modal>

      {/* Provider Modal */}
      <Modal visible={providerModalVisible} transparent animationType="slide">
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>Parkplatz anbieten</Text>
            <TextInput
              style={styles.input}
              placeholder="Titel (z.B. Privatparkplatz)"
              value={newTitle}
              onChangeText={setNewTitle}
            />
            <TextInput
              style={styles.input}
              placeholder="Preis pro Stunde (€)"
              keyboardType="decimal-pad"
              value={newPrice}
              onChangeText={setNewPrice}
            />
            <Button title="Speichern" onPress={addNewParking} />
            <Button title="Abbrechen" onPress={() => setProviderModalVisible(false)} />
          </View>
        </View>
      </Modal>

      {/* Show booking modal when parking selected */}
      {selectedParking && !bookingModalVisible && (
        <View style={styles.bookingBar}>
          <Text>{selectedParking.title} — {selectedParking.price} €/h</Text>
          <TouchableOpacity
            style={styles.bookButton}
            onPress={() => setBookingModalVisible(true)}
          >
            <Text style={{ color: 'white' }}>Jetzt buchen</Text>
          </TouchableOpacity>
          <TouchableOpacity
            style={styles.closeButton}
            onPress={() => setSelectedParking(null)}
          >
            <Text style={{ color: 'black', fontWeight: 'bold' }}>✕</Text>
          </TouchableOpacity>
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  map: { flex: 1 },
  marker: {
    backgroundColor: '#2ecc71',
    paddingHorizontal: 6,
    paddingVertical: 2,
    borderRadius: 4,
  },
  bookingBar: {
    position: 'absolute',
    bottom: 20,
    left: 20,
    right: 20,
    backgroundColor: 'white',
    borderRadius: 10,
    padding: 15,
    flexDirection: 'row',
    alignItems: 'center',
    elevation: 5,
    shadowColor: '#000',
    shadowOpacity: 0.3,
    shadowRadius: 6,
    shadowOffset: { width: 0, height: 2 },
  },
  bookButton: {
    backgroundColor: '#27ae60',
    paddingHorizontal: 15,
    paddingVertical: 8,
    borderRadius: 8,
    marginLeft: 'auto',
  },
  closeButton: {
    marginLeft: 10,
  },
  modalOverlay: {
    flex: 1,
    backgroundColor: '#00000099',
    justifyContent: 'center',
    padding: 20,
  },
  modalContent: {
    backgroundColor: 'white',
    borderRadius: 10,
    padding: 20,
  },
  modalTitle: {
    fontSize: 20,
    marginBottom: 15,
    fontWeight: 'bold',
  },
  input: {
    borderWidth: 1,
    borderColor: '#ccc',
    padding: 10,
    marginBottom: 15,
    borderRadius: 6,
  },
});
